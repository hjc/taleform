resumeBeat returns the furthest touched beat when it is partial, and the first later planned, non-optional beat otherwise. The builder prints one fixed sentence for both outcomes.

export function resumeBeat(seq: Note[]): Note | undefined { const beats = seq.filter((n) => n.kind !== "act"); let furthest = -1; beats.forEach((n, i) => { if (touched(n)) furthest = i; }); if (furthest >= 0 && status(beats[furthest]!) === "partial") return beats[furthest]; return beats.find((n, i) => i > furthest && status(n) === "planned" && !truthy(n.fm.optional)); } if (resume && beats.has(resume)) { const rid = nid("resume"); node(rid, { type: "text", x: LX, y: pos.get(resume)!.y + 140, width: LW, height: 90, color: RESUME, text: "## ▶ RESUME HERE\nthe first planned beat after the furthest one played" }); edge(rid, nid("n", resume), null, RESUME); }
